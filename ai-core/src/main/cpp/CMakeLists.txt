cmake_minimum_required(VERSION 3.22)
project(nneonneo_threes LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Keep the vendored source pristine. The adapter polls budgets and overrides the
# upstream adaptive depth for complete iterative-deepening rounds.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/upstream/threes.cpp" ENGINE_SOURCE)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/upstream/threes.cpp" ENGINE_SHA256)
if(NOT ENGINE_SHA256 STREQUAL "ee8f7ff5ff054e57f9bd37e7f92dbe64f26001cd59280b4aed4a68d19547f436")
    message(FATAL_ERROR "Vendored threes.cpp no longer matches the pinned upstream commit")
endif()
set(POLL_ANCHOR "static float score_move_node(eval_state &state, board_t board, deck_t deck, float cprob) {\n")
set(DEPTH_ANCHOR "    res = _score_toplevel_move(state, board, deck, tileset, move);")
string(FIND "${ENGINE_SOURCE}" "${POLL_ANCHOR}" POLL_POSITION)
string(FIND "${ENGINE_SOURCE}" "${DEPTH_ANCHOR}" DEPTH_POSITION)
if(POLL_POSITION EQUAL -1 OR DEPTH_POSITION EQUAL -1)
    message(FATAL_ERROR "Upstream source changed: re-audit the budget/depth hooks")
endif()
string(REPLACE "${POLL_ANCHOR}" "${POLL_ANCHOR}    threes_poll_search();\n" ENGINE_SOURCE "${ENGINE_SOURCE}")
string(REPLACE "${DEPTH_ANCHOR}" "    state.depth_limit = threes_search_depth(state.depth_limit);\n${DEPTH_ANCHOR}" ENGINE_SOURCE "${ENGINE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/threes_instrumented.cpp" "${ENGINE_SOURCE}")

add_library(nneonneo_threes SHARED nneonneo_jni.cpp)
target_include_directories(nneonneo_threes PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/upstream" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(nneonneo_threes PRIVATE -O3)
set_target_properties(nneonneo_threes PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

if(NOT ANDROID)
    if(NOT THREES_HOST_JAVA_HOME)
        message(FATAL_ERROR "Set THREES_HOST_JAVA_HOME to a JDK containing include/jni.h")
    endif()
    target_include_directories(nneonneo_threes PRIVATE "${THREES_HOST_JAVA_HOME}/include")
    if(WIN32)
        target_include_directories(nneonneo_threes PRIVATE "${THREES_HOST_JAVA_HOME}/include/win32")
        target_compile_definitions(nneonneo_threes PRIVATE _WINDLL NOMINMAX)
        target_link_options(nneonneo_threes PRIVATE -static)
        set_target_properties(nneonneo_threes PROPERTIES PREFIX "")
    elseif(APPLE)
        target_include_directories(nneonneo_threes PRIVATE "${THREES_HOST_JAVA_HOME}/include/darwin")
    else()
        target_include_directories(nneonneo_threes PRIVATE "${THREES_HOST_JAVA_HOME}/include/linux")
    endif()
    # A second, completely uninstrumented copy is an independent oracle for tests.
    target_sources(nneonneo_threes PRIVATE upstream_oracle.cpp)
    target_compile_definitions(nneonneo_threes PRIVATE THREES_ORACLE_AVAILABLE)
endif()
